Option A — Stream from a relay server

Browsers can't speak RTMP. Run the tiny relay below (or use OBS with the credentials above), then send the file here and it will be pushed live via FFmpeg.

Relay server code (Node + FFmpeg)

Save as server.js, then npm i express multer cors and node server.js. Requires ffmpeg on PATH. Host it somewhere with good upload bandwidth. 

// server.js
const express = require('express');
const multer  = require('multer');
const cors    = require('cors');
const { spawn } = require('child_process');
const fs = require('fs'), os = require('os'), path = require('path');

const app = express();
app.use(cors());

const upload = multer({ dest: path.join(os.tmpdir(), 'golive') });
const jobs = new Map();

app.post('/stream', upload.single('file'), (req, res) => {
  const { rtmpUrl, streamKey, loop } = req.body;
  if (!req.file || !rtmpUrl || !streamKey)
    return res.status(400).json({ error: 'missing file / rtmpUrl / streamKey' });

  const args = [];
  if (loop === '1') args.push('-stream_loop', '-1');
  args.push(
    '-re', '-i', req.file.path,
    '-c:v', 'libx264', '-preset', 'veryfast', '-tune', 'zerolatency',
    '-b:v', '4500k', '-maxrate', '4500k', '-bufsize', '9000k',
    '-pix_fmt', 'yuv420p', '-g', '60', '-r', '30',
    '-c:a', 'aac', '-b:a', '128k', '-ar', '44100',
    '-f', 'flv', `${rtmpUrl}/${streamKey}`
  );

  const ff = spawn('ffmpeg', args);
  const id = req.file.filename;
  jobs.set(id, ff);

  ff.stderr.on('data', d => process.stdout.write('[ffmpeg] ' + d));
  ff.on('close', () => { fs.unlink(req.file.path, () => {}); jobs.delete(id); });

  res.json({ ok: true, jobId: id });
});

app.post('/stop/:id', (req, res) => {
  const ff = jobs.get(req.params.id);
  if (!ff) return res.status(404).json({ error: 'no such job' });
  ff.kill('SIGINT');
  res.json({ ok: true });
});

app.listen(8080, () => console.log('relay listening on :8080'));

Or skip the app entirely and use ffmpeg by hand:

ffmpeg -re -stream_loop -1 -i myvideo.mp4 \
  -c:v libx264 -preset veryfast -tune zerolatency \
  -b:v 4500k -maxrate 4500k -bufsize 9000k \
  -pix_fmt yuv420p -g 60 -r 30 \
  -c:a aac -b:a 128k -ar 44100 \
  -f flv "rtmp://a.rtmp.youtube.com/live2/YOUR_STREAM_KEY"
Reminder: remove -stream_loop -1 if you only want it played once.
